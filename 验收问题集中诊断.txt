    for module_name in MODEL_MODULES:
        try:
            import_module(module_name)
            print(f"[模型已注册] {module_name}")
        except ModuleNotFoundError as exc:

def register_routers():
    for module_name in ROUTER_MODULES:
        try:
            module = importlib.import_module(module_name)
            router = getattr(module, "router", None)

            if router is None:
                router = getattr(module, "api_router", None)

                logger.warning("模块没有找到 router：%s", module_name)
                continue

            app.include_router(router)
            _loaded_routers.append(module_name)

                logger.warning("路由模块不存在，跳过：%s", module_name)
            else:
                logger.exception("路由依赖导入失败：%s", module_name)

def split_chapters(text):
    pattern = re.compile(
        r"(?m)^\s*(第[零〇一二三四五六七八九十百千万两\d]+[章节回卷集].*)$"
        r"|^\s*((?:Chapter|CHAPTER)\s+\d+.*)$"
    )
    matches = list(pattern.finditer(text))
    if not matches:
        content = text.strip()
        return [("正文", content)] if content else []

    result = []
    if text[:matches[0].start()].strip():
        result.append(("序章", text[:matches[0].start()].strip()))

    for index, match in enumerate(matches):
        title = (match.group(1) or match.group(2) or "").strip()
        start = match.end()
        end = matches[index + 1].start() if index + 1 < len(matches) else len(text)
        content = text[start:end].strip()
        result.append((title or f"第{index + 1}章", content))

    return result

@router.post("/novels/{novel_id}/import-txt")
async def import_txt(
    novel_id: int,
    file: UploadFile = File(...),
    db: Session = Depends(get_db),
):
    item = db.query(Novel).filter(Novel.id == novel_id).first()
    if not item:
        raise HTTPException(404, "小说不存在")

    filename = file.filename or "novel.txt"
    if Path(filename).suffix.lower() != ".txt":
        raise HTTPException(400, "目前仅支持 TXT 文件")

    raw = await file.read()
    if len(raw) > 100 * 1024 * 1024:
        raise HTTPException(413, "TXT 文件不能超过 100 MB")

    text = None
    for encoding in ("utf-8-sig", "utf-8", "gb18030"):
        try:
            text = raw.decode(encoding)
            break
        except UnicodeDecodeError:
            continue

    if text is None:
        raise HTTPException(400, "文件编码无法识别，请转换为 UTF-8")

    parts = split_chapters(text)
    if not parts:
        raise HTTPException(400, "文件内容为空")

    item.source_text = text
    item.title = item.title or Path(filename).stem

    for old in list(item.chapters):
        db.delete(old)

    for number, (title, content) in enumerate(parts, start=1):
        db.add(Chapter(
            novel_id=item.id,
            chapter_number=number,
            title=title[:300],
            content=content,
            word_count=len(re.sub(r"\s+", "", content)),
        ))

    db.commit()
    db.refresh(item)
    return {
        "success": True,
        "filename": filename,
        "chapter_count": len(parts),
        "word_count": len(re.sub(r"\s+", "", text)),
        "chapters": [chapter_dict(x) for x in item.chapters],
    }

@router.post("/novels/{novel_id}/analyze")
def analyze_novel(novel_id: int, db: Session = Depends(get_db)):
    item = db.query(Novel).filter(Novel.id == novel_id).first()
    if not item:
        raise HTTPException(404, "小说不存在")
    if not item.chapters:
        raise HTTPException(400, "请先导入小说或创建章节")

    # 本地规则初步提取：这是基础识别，不冒充外部 AI 模型分析。
    full_text = "\n".join(c.content for c in item.chapters)
    candidates = []

    quoted_names = re.findall(r"[“「『]([^”」』]{2,12})[”」』]", full_text)
    for name in dict.fromkeys(quoted_names):
        candidates.append(("character", name, "从文本引号内容中提取的待核对名称"))

    scene_words = ("大厅", "宫殿", "山谷", "森林", "城池", "村庄", "洞穴", "学校", "医院")
    for word in scene_words:
        if word in full_text:
            candidates.append(("scene", word, "文本中出现的场景线索"))

    prop_words = ("宝剑", "长剑", "法宝", "玉佩", "戒指", "卷轴", "手机", "钥匙")
    for word in prop_words:
        if word in full_text:
            candidates.append(("prop", word, "文本中出现的道具线索"))

    event_words = ("突然", "于是", "最终", "随后", "大战", "相遇", "逃跑", "发现")
    for word in event_words:
        count = full_text.count(word)
        if count:
            candidates.append(("event", word, f"剧情事件线索，出现 {count} 次"))

    for old in db.query(AnalysisItem).filter(
        AnalysisItem.novel_id == novel_id,
        AnalysisItem.confirmed == False
    ).all():
        db.delete(old)

    existing = set()
    created = []
    for category, name, description in candidates:
        key = (category, name)
        if key in existing:
            continue
        existing.add(key)
        row = AnalysisItem(
            novel_id=novel_id,
            category=category,
            name=name,
            description=description,
            data={"source": "local_rule", "needs_review": True},
        )
        db.add(row)
        created.append(row)

    db.commit()
    return {
        "success": True,
        "mode": "local_rule_preliminary",
        "message": "基础文本线索提取完成。接入 AI 模型后可进行深度语义分析。",
        "items": [analysis_dict(x) for x in created],
        "total": len(created),
    }

def get_cipher():
    if not KEY_FILE.exists():
        KEY_FILE.write_bytes(Fernet.generate_key())
    return Fernet(KEY_FILE.read_bytes())

def encrypt_key(value):
    if not value:
        return ""
    return get_cipher().encrypt(value.encode("utf-8")).decode("utf-8")

def decrypt_key(value):
    if not value:
        return ""
    try:
        return get_cipher().decrypt(value.encode("utf-8")).decode("utf-8")
    except InvalidToken:
        raise HTTPException(500, "API密钥解密失败，请检查本地密钥文件")

def ai_chat(payload: ChatInput, db: Session = Depends(get_db)):
    settings = runtime(db)
    config = db.query(AIProviderConfig).filter(
        AIProviderConfig.provider == payload.provider.lower()
    ).first()

    if not settings.network_enabled:
        raise HTTPException(403, "网络调用总开关已关闭")
    if not config or not config.enabled:
        raise HTTPException(400, "模型不存在或尚未启用")

        raise HTTPException(400, "素材分类无效")

    suffix = Path(original).suffix.lower()
    stored = f"{uuid.uuid4().hex}{suffix}"
    destination = ASSET_DIR / stored

    size = 0
    try:
        with destination.open("wb") as output:
            while True:
                chunk = await file.read(1024 * 1024)
                if not chunk:
                    break
                size += len(chunk)
                if size > 2 * 1024 * 1024 * 1024:
                    raise HTTPException(413, "单个素材文件不能超过 2 GB")
                output.write(chunk)

        item = MediaAsset(
            filename=original,
            stored_name=stored,
            category=category,
            mime_type=file.content_type or "",
            file_size=size,
            file_path=str(destination),
            description=description,
        )
        db.add(item)
        db.commit()
        db.refresh(item)
        return asset_dict(item)
    except Exception:
        if destination.exists():
            destination.unlink()
        raise

def decrypt_key(encrypted):
    if not encrypted:
        return ""
    key_file = ROOT / "storage" / "secrets" / "local.key"
    if not key_file.exists():
        raise HTTPException(400, "本地加密密钥不存在，无法读取服务商 API Key")
    try:
        key = key_file.read_bytes()
        return Fernet(key).decrypt(encrypted.encode()).decode()
    except (InvalidToken, ValueError):
        raise HTTPException(400, "API Key 解密失败，请重新保存服务商配置")

@router.post("/tasks/{task_id}/run")
def run_task(task_id: int, confirm_paid: bool = False, db: Session = Depends(get_db)):
    task = db.query(ImageTask).filter(ImageTask.id == task_id).first()
    if not task:
        raise HTTPException(404, "图片任务不存在")
    if task.status == "completed":
        return {"success": True, "task_id": task.id, "status": task.status, "result_path": task.result_path}
    if task.status == "running":
        raise HTTPException(409, "任务正在执行")

    runtime = get_runtime(db)
    if not runtime["network_enabled"]:
        raise HTTPException(403, "网络调用总开关已关闭，请先到 AI 管理中心开启")
    
    provider = db.query(AIProviderConfig).filter(
        AIProviderConfig.provider == task.provider
    ).first()
    if not provider or not provider.enabled:
        raise HTTPException(400, f"{task.provider} 服务商尚未配置或未启用")
    if not provider.encrypted_api_key:
        raise HTTPException(400, "服务商尚未保存 API Key")
    if bool(provider.paid) and (not runtime["paid_api_allowed"] or not confirm_paid):
        raise HTTPException(403, "该服务商可能产生费用。请开启付费授权，并在本次任务中明确确认")

    task.status = "running"
    task.error = ""
    db.commit()

    try:
        api_key = decrypt_key(provider.encrypted_api_key)
        base_url = (provider.base_url or "").rstrip("/")
        if not base_url:
            raise ValueError("服务商 Base URL 未配置")

        # 豆包 Ark 图片生成接口采用 OpenAI 风格的 images/generations 请求格式。
        endpoint = base_url if base_url.endswith("/images/generations") else base_url + "/images/generations"
        body = {
            "model": provider.model,
            "prompt": task.prompt,
            "size": task.size,
            "response_format": "url",
        }
        request = urllib.request.Request(
            endpoint,
            data=json.dumps(body, ensure_ascii=False).encode("utf-8"),
            headers={
                "Authorization": f"Bearer {api_key}",
                "Content-Type": "application/json",
            },
            method="POST",
        )
        with urllib.request.urlopen(request, timeout=300) as response:
            result = json.loads(response.read().decode("utf-8"))

        data = result.get("data") or []
        if not data:
            raise ValueError("图片服务没有返回图片数据：" + json.dumps(result, ensure_ascii=False)[:1000])

        first = data[0]
        image_url = first.get("url", "")
        b64_data = first.get("b64_json", "")

        if b64_data:
            image_bytes = base64.b64decode(b64_data)
            filename = f"image_task_{task.id}.png"
            output = ASSET_DIR / filename
            output.write_bytes(image_bytes)
            task.result_path = str(output)
        elif image_url:
            task.result_url = image_url
            download_request = urllib.request.Request(image_url)
            with urllib.request.urlopen(download_request, timeout=300) as response:
                image_bytes = response.read()
            filename = f"image_task_{task.id}.png"
            output = ASSET_DIR / filename
            output.write_bytes(image_bytes)
            task.result_path = str(output)
        else:
            raise ValueError("服务商返回结果中没有图片 URL 或 b64_json")

        task.model = provider.model
        task.status = "completed"
        task.finished_at = datetime.utcnow()

        if task.asset_id:
            asset = db.query(VisualAsset).filter(VisualAsset.id == task.asset_id).first()
            if asset:
                asset.image_path = task.result_path
                asset.updated_at = datetime.utcnow()

        db.commit()
        return {
            "success": True,
            "task_id": task.id,
            "status": task.status,
            "result_path": task.result_path,
            "result_url": task.result_url,
        }

    except urllib.error.HTTPError as exc:
        detail = exc.read().decode("utf-8", errors="replace")[:2000]
        task.status = "failed"
        task.error = f"图片服务 HTTP {exc.code}: {detail}"
        db.commit()
        raise HTTPException(502, task.error)
    except Exception as exc:
        task.status = "failed"
        task.error = str(exc)[:2000]
        db.commit()
        raise HTTPException(500, task.error)

def serialize_model(obj):
    result = {}
    for column in sa_inspect(obj.__class__).mapper.column_attrs:
        value = getattr(obj, column.key)
        if isinstance(value, datetime):
            value = value.isoformat()
        result[column.key] = value
    return result


def chapter_value(chapter, candidates, default=""):
    for name in candidates:
        if hasattr(chapter, name):
            value = getattr(chapter, name)
            if value is not None and value != "":
                return value
    return default


def get_chapter_content(chapter):
    return str(chapter_value(
        chapter,
        ["content", "body", "text", "chapter_content", "original_text"],
        ""
    ))


def get_chapter_title(chapter):
    return str(chapter_value(
        chapter,
        ["title", "name", "chapter_title"],
        f"第{chapter.id}章"
    ))


def get_novel_chapters(db, novel_id):
    chapters = db.query(Chapter).all()
    result = []
    for chapter in chapters:
        chapter_novel_id = chapter_value(
            chapter, ["novel_id", "book_id"], None
        )
        if chapter_novel_id is not None and int(chapter_novel_id) == int(novel_id):
            result.append(chapter)

    result.sort(key=lambda x: chapter_value(
        x, ["chapter_number", "chapter_index", "order", "sort_order", "id"], x.id
    ))
    return result

    episode_number = 1
    for chapter in chapters:
        content = get_chapter_content(chapter)
        title = get_chapter_title(chapter)
        if not content.strip():
            continue

        # 用章节内容长度估算分集；此处是初步规划规则，不冒充 AI 剧情分析。
        split = len(content) > 9000
        parts = [content[:len(content)//2], content[len(content)//2:]] if split else [content]

        for part_index, part in enumerate(parts):
            suffix = f"（下）" if split and part_index == 1 else (f"（上）" if split else "")
            episode = EpisodePlan(
                project_id=project_id,
                episode_number=episode_number,
                title=f"{title}{suffix}",
                source_chapter_ids=[chapter.id],
                source_text=part,
                synopsis="待 AI 根据原著内容提炼剧情摘要",
                screenplay="待 AI 改编为影视剧本",
                duration_seconds=project.target_duration,
                status="planned",
                sort_order=episode_number,
            )
            db.add(episode)
            episode_number += 1

    project.status = "planned"
    project.updated_at = datetime.utcnow()
    db.commit()

    episodes = db.query(EpisodePlan).filter(
        EpisodePlan.project_id == project_id
    ).order_by(EpisodePlan.sort_order).all()

    return {
        "success": True,
        "message": "初步分集计划已建立。剧情摘要和剧本正文仍需 AI 改编。",
        "episode_count": len(episodes),
        "episodes": [serialize_model(x) for x in episodes],
    }

        raise HTTPException(400, "至少选择两集才能合并")
    episodes = db.query(EpisodePlan).filter(
        EpisodePlan.project_id == project_id,
        EpisodePlan.id.in_(payload.episode_ids)
    ).order_by(EpisodePlan.sort_order).all()
    if len(episodes) != len(set(payload.episode_ids)):
        raise HTTPException(400, "分集不存在或不属于该项目")

    first = episodes[0]
    first.title = payload.title.strip() or " + ".join(x.title for x in episodes)
    first.source_text = "\n\n".join(x.source_text or "" for x in episodes)
    first.source_chapter_ids = list(dict.fromkeys(
        chapter_id for x in episodes for chapter_id in (x.source_chapter_ids or [])
    ))
    first.duration_seconds = min(
        3600, sum(x.duration_seconds for x in episodes)
    )
    for item in episodes[1:]:
        db.query(StoryboardShot).filter(StoryboardShot.episode_id == item.id).update(
            {StoryboardShot.episode_id: first.id}
        )
        db.query(StoryAssetLink).filter(StoryAssetLink.episode_id == item.id).update(
            {StoryAssetLink.episode_id: first.id}
        )
        db.delete(item)
    db.commit()
    return {"success": True, "merged_episode_id": first.id}


@router.post("/episodes/{episode_id}/split")
def split_episode(episode_id: int, db: Session = Depends(get_db)):
    episode = db.query(EpisodePlan).filter(EpisodePlan.id == episode_id).first()
    if not episode:
        raise HTTPException(404, "分集不存在")
    text = episode.source_text or ""
    if len(text.strip()) < 2:
        raise HTTPException(400, "原始内容太短，无法拆分")
    midpoint = len(text) // 2
    split_at = text.find("\n", midpoint - 500, midpoint + 500)
    if split_at < 0:
        split_at = midpoint

    second = EpisodePlan(
        project_id=episode.project_id,
        episode_number=episode.episode_number + 1,
        title=episode.title + "（下）",
        source_chapter_ids=episode.source_chapter_ids or [],
        source_text=text[split_at:],
        synopsis="待 AI 提炼剧情摘要",
        screenplay="待 AI 改编剧本",
        duration_seconds=episode.duration_seconds,
        status="planned",
        sort_order=episode.sort_order + 1,
    )
    episode.title = episode.title + "（上）"
    episode.source_text = text[:split_at]
    db.add(second)
    db.flush()

    later = db.query(EpisodePlan).filter(
        EpisodePlan.project_id == episode.project_id,
        EpisodePlan.id != episode.id,
        EpisodePlan.sort_order >= second.sort_order
    ).all()
    for item in later:
        item.sort_order += 1
        item.episode_number += 1

    db.commit()
    return {"success": True, "new_episode": serialize_model(second)}

def get_provider(db, provider_name, confirm_paid=False):
    config = db.query(AIProviderConfig).filter(
        AIProviderConfig.provider == provider_name
    ).first()
    if not config:
        raise HTTPException(404, f"AI 服务商 {provider_name} 尚未配置")

    if not getattr(config, "enabled", False):
        raise HTTPException(400, "该 AI 服务商尚未启用")

    runtime = db.query(AIRuntimeSetting).first()
    if not runtime or not getattr(runtime, "network_enabled", False):
        raise HTTPException(403, "全局网络调用开关已关闭")

    paid = bool(getattr(config, "paid", True))
    if paid:
        if not getattr(runtime, "paid_api_allowed", False):
            raise HTTPException(403, "付费 API 总授权未开启")
        if not confirm_paid:
            raise HTTPException(403, "此操作可能产生费用，必须明确确认后才能调用")

    encrypted = (
        getattr(config, "api_key_encrypted", None)
        or getattr(config, "encrypted_api_key", None)
    )
    if not encrypted:
        raise HTTPException(400, "该服务商尚未保存 API Key")

    base_url = getattr(config, "base_url", None)
    if not base_url:
        raise HTTPException(400, "该服务商尚未设置 API Base URL")

    try:
        api_key = decrypt_api_key(encrypted)
    except Exception as exc:
        raise HTTPException(500, str(exc))

    return config, api_key, base_url.rstrip("/")


def value_at(data, dotted_path):
    current = data
    for part in dotted_path.split("."):
        if isinstance(current, dict):
            current = current.get(part)
        elif isinstance(current, list) and part.isdigit():
            current = current[int(part)]
        else:
            return None
        if current is None:
            return None
    return current

    config, key, base = get_provider(db, data.provider, data.confirm_paid)

    payload = {
        "model": data.model,
        "input": data.text,
        "voice": data.voice,
        "response_format": data.response_format,
        "speed": data.speed
    }
    if data.instructions:
        payload["instructions"] = data.instructions

    filename = f"speech_{uuid.uuid4().hex}.{data.response_format}"
    destination = AUDIO_DIR / filename
    import urllib.request
    import urllib.error

    request = urllib.request.Request(
        base + "/audio/speech",
        data=json.dumps(payload, ensure_ascii=False).encode("utf-8"),
        headers={
            "Authorization": f"Bearer {key}",
            "Content-Type": "application/json"
        },
        method="POST"
    )
    try:
        with urllib.request.urlopen(request, timeout=300) as response:
            with open(destination, "wb") as output:
                while True:
                    chunk = response.read(1024 * 1024)
                    if not chunk:
                        break
                    output.write(chunk)

        media = MediaFile(
            name=filename, category="audio", file_path=str(destination),
            mime_type="audio/" + data.response_format,
            file_size=destination.stat().st_size
        )
        task = MediaTask(
            task_type="speech", provider=data.provider, model=data.model,
            status="completed", prompt=data.text, result_path=str(destination)
        )
        db.add(media)
        db.add(task)
        db.commit()
        return {"status": "completed", "file": filename,
                "path": str(destination), "size": destination.stat().st_size}
    except urllib.error.HTTPError as exc:
        detail = exc.read().decode("utf-8", errors="replace")
        raise HTTPException(502, f"语音 API 返回 HTTP {exc.code}: {detail[:1500]}")
    except Exception as exc:
        raise HTTPException(502, f"语音生成失败：{exc}")


@router.post("/files/upload")
async def upload_media(file: UploadFile = File(...), category: str = "video",
                       db: Session = Depends(get_db)):
    allowed = {"video": VIDEO_DIR, "audio": AUDIO_DIR, "image": STORAGE / "images"}
    if category not in allowed:
        raise HTTPException(400, "不支持的素材类别")

    folder = allowed[category]
    folder.mkdir(parents=True, exist_ok=True)
    suffix = Path(file.filename or "upload.bin").suffix[:20]
    filename = f"{uuid.uuid4().hex}{suffix}"
    destination = folder / filename
    size = 0
    limit = 2 * 1024 * 1024 * 1024

    try:
        with open(destination, "wb") as output:
            while True:
                chunk = await file.read(1024 * 1024)
                if not chunk:
                    break
                size += len(chunk)
                if size > limit:
                    raise HTTPException(413, "单个文件不能超过 2GB")
                output.write(chunk)

        media = MediaFile(
            name=file.filename or filename, category=category,
            file_path=str(destination), mime_type=file.content_type,
            file_size=size
        )
        db.add(media)
        db.commit()
        db.refresh(media)
        return {"id": media.id, "name": media.name, "category": category,
                "size": size, "path": media.file_path}
    except Exception:
        if destination.exists():
            destination.unlink()
        raise

def srt_time(seconds):
    ms = int(round(seconds * 1000))
    hours, ms = divmod(ms, 3600000)
    minutes, ms = divmod(ms, 60000)
    secs, ms = divmod(ms, 1000)
    return f"{hours:02}:{minutes:02}:{secs:02},{ms:03}"


@router.post("/subtitles")
def create_subtitle(data: SubtitleInput, db: Session = Depends(get_db)):
    if not data.cues:
        raise HTTPException(400, "字幕内容不能为空")
    lines = []
    for index, cue in enumerate(data.cues, 1):
        if cue.end <= cue.start:
            raise HTTPException(400, f"第 {index} 条字幕结束时间必须大于开始时间")
        lines.extend([
            str(index),
            f"{srt_time(cue.start)} --> {srt_time(cue.end)}",
            cue.text,
            ""
        ])
    filename = f"subtitle_{uuid.uuid4().hex}.srt"
    destination = SUBTITLE_DIR / filename
    destination.write_text("\n".join(lines), encoding="utf-8-sig")

    track = SubtitleTrack(name=data.name, file_path=str(destination), format="srt")
    db.add(track)
    db.commit()
    db.refresh(track)
    return {"id": track.id, "name": track.name, "path": track.file_path}

            )

                     "-map", "[mix]", "-shortest"]
        else:

        if subtitle:
            escaped = str(subtitle).replace("\\", "/").replace(":", "\\:")
            args += ["-vf", f"subtitles='{escaped}'"]

            "-c:v", "libx264", "-preset", "medium", "-crf", "20",
            "-c:a", "aac", "-movflags", "+faststart", str(output)
        ]
        run_ffmpeg(args)
        return {"status": "completed", "output": str(output),
                "size": output.stat().st_size}

router = APIRouter(prefix="/api/video-editor", tags=["AI 视频剪辑"])


class CreateEditInput(BaseModel):
    title: str = "视频剪辑任务"
    provider: str
    model: str
    instruction: str = Field(min_length=2, max_length=12000)
    source_paths: list[str] = Field(min_length=1, max_length=30)
    confirm_paid: bool = False


class ExecuteEditInput(BaseModel):
    subtitle_path: Optional[str] = None
    audio_path: Optional[str] = None


@router.post("/jobs")
def create_edit_job(data: CreateEditInput, db: Session = Depends(get_db)):
    try:
        sources = []
        for raw in data.source_paths:
            path = safe_storage_file(raw)
            sources.append(str(path))

        job = VideoEditJob(
            title=data.title,
            provider=data.provider,
            model=data.model,
            instruction=data.instruction,
            status="analyzing",
            source_files=sources
        )
        db.add(job)
        db.commit()
        db.refresh(job)

        plan, raw_response = call_ai_for_edit_plan(
            db, data.provider, data.model,
            data.instruction, sources, data.confirm_paid
        )
        validated = validate_plan(plan, sources)

        job.edit_plan = validated
        job.title = validated["title"] or data.title
        job.status = "planned"
        db.commit()

        return {
            "id": job.id,
            "status": job.status,
            "plan": validated
        }
    except Exception as exc:
        db.rollback()
        raise HTTPException(502, f"AI 剪辑方案生成失败：{exc}")

@router.post("/jobs/{job_id}/execute")
def execute_job(job_id: int, data: ExecuteEditInput,
                db: Session = Depends(get_db)):
    job = db.query(VideoEditJob).filter(
        VideoEditJob.id == job_id
    ).first()
    if not job:
        raise HTTPException(404, "剪辑任务不存在")

    if job.status != "planned" or not job.edit_plan:
        raise HTTPException(400, "任务尚未生成有效剪辑方案")

    try:
        job.status = "rendering"
        db.commit()

        output, size = execute_edit_plan(
            job.edit_plan,
            subtitle_path=data.subtitle_path,
            audio_path=data.audio_path
        )

        job.output_path = output
        job.status = "completed"
        job.error = None

        media = MediaFile(
            name=Path(output).name,
            category="video",
            file_path=output,
            mime_type="video/mp4",
            file_size=size
        )
        db.add(media)
        db.commit()

        return {
            "id": job.id,
            "status": job.status,
            "output_path": output,
            "size": size
        }
    except Exception as exc:
        job.status = "failed"
        job.error = str(exc)
        db.commit()
        raise HTTPException(500, f"视频剪辑执行失败：{exc}")

const { app, BrowserWindow, dialog } = require("electron");
const { spawn } = require("child_process");
const path = require("path");
const fs = require("fs");
const net = require("net");

let mainWindow = null;
let apiProcess = null;
let webProcess = null;
let quitting = false;

const API_PORT = 8000;
const WEB_PORT = 3000;

function waitForPort(port, timeout = 120000) {
  return new Promise((resolve, reject) => {
    const start = Date.now();

    function check() {
      const socket = new net.Socket();
      socket.setTimeout(1500);

      socket.once("connect", () => {
        socket.destroy();
        resolve(true);
      });

      socket.once("error", () => {
        socket.destroy();
        if (Date.now() - start > timeout) {
          reject(new Error(`等待端口 ${port} 超时`));
        } else {
          setTimeout(check, 1000);
        }
      });

      socket.once("timeout", () => {
        socket.destroy();
        if (Date.now() - start > timeout) {
          reject(new Error(`等待端口 ${port} 超时`));
        } else {
          setTimeout(check, 1000);
        }
      });

      socket.connect(port, "127.0.0.1");
    }

    check();
  });
}

function startProductionServices() {
  const resources = process.resourcesPath;
  const apiExe = path.join(resources, "api", "ai-comic-api.exe");
  const webServer = path.join(resources, "web", "server.js");
  const dataDir = app.getPath("userData");

  fs.mkdirSync(dataDir, { recursive: true });

  if (!fs.existsSync(apiExe)) {
    throw new Error(`找不到后端程序：${apiExe}`);
  }

  if (!fs.existsSync(webServer)) {
    throw new Error(`找不到前端程序：${webServer}`);
  }

  apiProcess = spawn(apiExe, [], {
    cwd: resources,
    windowsHide: true,
    env: {
      ...process.env,
      APP_DATA_DIR: dataDir,
      PYTHONUNBUFFERED: "1",
      FFMPEG_PATH: process.env.FFMPEG_PATH || "ffmpeg",
      FFPROBE_PATH: process.env.FFPROBE_PATH || "ffprobe"
    }
  });

  apiProcess.on("error", error => {
    console.error("后端启动失败：", error);
  });

  apiProcess.stdout?.on("data", data => {
    console.log("[API]", data.toString());
  });

  apiProcess.stderr?.on("data", data => {
    console.error("[API]", data.toString());
  });

  webProcess = spawn(process.execPath, [webServer], {
    cwd: path.join(resources, "web"),
    windowsHide: true,
    env: {
      ...process.env,
      ELECTRON_RUN_AS_NODE: "1",
      PORT: String(WEB_PORT),
      HOSTNAME: "127.0.0.1",
      NEXT_PUBLIC_API_BASE_URL: "http://127.0.0.1:8000"
    }
  });

  webProcess.on("error", error => {
    console.error("前端启动失败：", error);
  });

  webProcess.stdout?.on("data", data => {
    console.log("[WEB]", data.toString());
  });

  webProcess.stderr?.on("data", data => {
    console.error("[WEB]", data.toString());
  });
}

async function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1600,
    height: 950,
    minWidth: 1100,
    minHeight: 700,
    show: false,
    autoHideMenuBar: true,
    backgroundColor: "#10091f",
    webPreferences: {
      preload: path.join(__dirname, "preload.cjs"),
      contextIsolation: true,
      nodeIntegration: false
    }
  });

  mainWindow.once("ready-to-show", () => {
    mainWindow.show();
  });

  mainWindow.on("closed", () => {
    mainWindow = null;
  });

  if (app.isPackaged) {
    startProductionServices();

    await waitForPort(API_PORT);
    await waitForPort(WEB_PORT);

    await mainWindow.loadURL("http://127.0.0.1:3000");
  } else {
    await mainWindow.loadURL("http://127.0.0.1:3000");
  }
}

function stopServices() {
  for (const child of [webProcess, apiProcess]) {
    if (child && !child.killed) {
      try {
        child.kill();
      } catch (error) {
        console.error(error);
      }
    }
  }
}

app.whenReady().then(async () => {

    await createWindow();

    dialog.showErrorBox(
      "AI漫剧制作大师启动失败",
      String(error.message || error)
    );
    app.quit();
  }
});

app.on("before-quit", () => {
  quitting = true;
  stopServices();
});

app.on("window-all-closed", () => {
  if (process.platform !== "darwin") {
    app.quit();
  }
});

app.on("activate", () => {
  if (BrowserWindow.getAllWindows().length === 0 && !quitting) {
    createWindow();
  }
});
