def split_chapters(text):
    pattern = re.compile(
        r"(?m)^\s*(第[零〇一二三四五六七八九十百千万两\d]+[章节回卷集].*)$"
        r"|^\s*((?:Chapter|CHAPTER)\s+\d+.*)$"
    )
    matches = list(pattern.finditer(text))
    if not matches:
        content = text.strip()
        return [("正文", content)] if content else []

    result = []
    if text[:matches[0].start()].strip():
        result.append(("序章", text[:matches[0].start()].strip()))

    for index, match in enumerate(matches):
        title = (match.group(1) or match.group(2) or "").strip()
        start = match.end()
        end = matches[index + 1].start() if index + 1 < len(matches) else len(text)
        content = text[start:end].strip()
        result.append((title or f"第{index + 1}章", content))

    return result

@router.post("/novels/{novel_id}/import-txt")
async def import_txt(
    novel_id: int,
    file: UploadFile = File(...),
    db: Session = Depends(get_db),
):
    item = db.query(Novel).filter(Novel.id == novel_id).first()
    if not item:
        raise HTTPException(404, "小说不存在")

    filename = file.filename or "novel.txt"
    if Path(filename).suffix.lower() != ".txt":
        raise HTTPException(400, "目前仅支持 TXT 文件")

    raw = await file.read()
    if len(raw) > 100 * 1024 * 1024:
        raise HTTPException(413, "TXT 文件不能超过 100 MB")

    text = None
    for encoding in ("utf-8-sig", "utf-8", "gb18030"):
        try:
            text = raw.decode(encoding)
            break
        except UnicodeDecodeError:
            continue

    if text is None:
        raise HTTPException(400, "文件编码无法识别，请转换为 UTF-8")

    parts = split_chapters(text)
    if not parts:
        raise HTTPException(400, "文件内容为空")

    item.source_text = text
    item.title = item.title or Path(filename).stem

    for old in list(item.chapters):
        db.delete(old)

    for number, (title, content) in enumerate(parts, start=1):
        db.add(Chapter(
            novel_id=item.id,
            chapter_number=number,
            title=title[:300],
            content=content,
            word_count=len(re.sub(r"\s+", "", content)),
        ))

    db.commit()
    db.refresh(item)
    return {
        "success": True,
        "filename": filename,
        "chapter_count": len(parts),
        "word_count": len(re.sub(r"\s+", "", text)),
        "chapters": [chapter_dict(x) for x in item.chapters],
    }

@router.post("/novels/{novel_id}/analyze")
def analyze_novel(novel_id: int, db: Session = Depends(get_db)):
    item = db.query(Novel).filter(Novel.id == novel_id).first()
    if not item:
        raise HTTPException(404, "小说不存在")
    if not item.chapters:
        raise HTTPException(400, "请先导入小说或创建章节")

    # 本地规则初步提取：这是基础识别，不冒充外部 AI 模型分析。
    full_text = "\n".join(c.content for c in item.chapters)
    candidates = []

    quoted_names = re.findall(r"[“「『]([^”」』]{2,12})[”」』]", full_text)
    for name in dict.fromkeys(quoted_names):
        candidates.append(("character", name, "从文本引号内容中提取的待核对名称"))

    scene_words = ("大厅", "宫殿", "山谷", "森林", "城池", "村庄", "洞穴", "学校", "医院")
    for word in scene_words:
        if word in full_text:
            candidates.append(("scene", word, "文本中出现的场景线索"))

    prop_words = ("宝剑", "长剑", "法宝", "玉佩", "戒指", "卷轴", "手机", "钥匙")
    for word in prop_words:
        if word in full_text:
            candidates.append(("prop", word, "文本中出现的道具线索"))

    event_words = ("突然", "于是", "最终", "随后", "大战", "相遇", "逃跑", "发现")
    for word in event_words:
        count = full_text.count(word)
        if count:
            candidates.append(("event", word, f"剧情事件线索，出现 {count} 次"))

    for old in db.query(AnalysisItem).filter(
        AnalysisItem.novel_id == novel_id,
        AnalysisItem.confirmed == False
    ).all():
        db.delete(old)

    existing = set()
    created = []
    for category, name, description in candidates:
        key = (category, name)
        if key in existing:
            continue
        existing.add(key)
        row = AnalysisItem(
            novel_id=novel_id,
            category=category,
            name=name,
            description=description,
            data={"source": "local_rule", "needs_review": True},
        )
        db.add(row)
        created.append(row)

    db.commit()
    return {
        "success": True,
        "mode": "local_rule_preliminary",
        "message": "基础文本线索提取完成。接入 AI 模型后可进行深度语义分析。",
        "items": [analysis_dict(x) for x in created],
        "total": len(created),
    }

def get_provider(db, provider_name, confirm_paid=False):
    config = db.query(AIProviderConfig).filter(
        AIProviderConfig.provider == provider_name
    ).first()
    if not config:
        raise HTTPException(404, f"AI 服务商 {provider_name} 尚未配置")

    if not getattr(config, "enabled", False):
        raise HTTPException(400, "该 AI 服务商尚未启用")

    runtime = db.query(AIRuntimeSetting).first()
    if not runtime or not getattr(runtime, "network_enabled", False):
        raise HTTPException(403, "全局网络调用开关已关闭")

    paid = bool(getattr(config, "paid", True))
    if paid:
        if not getattr(runtime, "paid_api_allowed", False):
            raise HTTPException(403, "付费 API 总授权未开启")
        if not confirm_paid:
            raise HTTPException(403, "此操作可能产生费用，必须明确确认后才能调用")

    encrypted = (
        getattr(config, "api_key_encrypted", None)
        or getattr(config, "encrypted_api_key", None)
    )
    if not encrypted:
        raise HTTPException(400, "该服务商尚未保存 API Key")

    base_url = getattr(config, "base_url", None)
    if not base_url:
        raise HTTPException(400, "该服务商尚未设置 API Base URL")

    try:
        api_key = decrypt_api_key(encrypted)
    except Exception as exc:
        raise HTTPException(500, str(exc))

    return config, api_key, base_url.rstrip("/")


def value_at(data, dotted_path):
    current = data
    for part in dotted_path.split("."):
        if isinstance(current, dict):
            current = current.get(part)
        elif isinstance(current, list) and part.isdigit():
            current = current[int(part)]
        else:
            return None
        if current is None:
            return None
    return current

@router.post("/video/tasks")
def create_video_task(data: VideoTaskInput, db: Session = Depends(get_db)):
    config, key, base = get_provider(db, data.provider, data.confirm_paid)

    payload = {
        "model": data.model,
        "prompt": data.prompt,
        "duration": data.duration,
        "aspect_ratio": data.aspect_ratio,
        **data.extra_payload
    }
    path = "/" + data.create_path.lstrip("/")
    try:
        result = http_json(base + path, key, body=payload, timeout=300)
        remote_id = value_at(result, data.task_id_field)
        if not remote_id:
            raise RuntimeError(
                "视频服务商没有返回任务编号。请按该服务商文档配置请求路径和返回字段。"
            )

        task = MediaTask(
            task_type="video",
            provider=data.provider,
            model=data.model,
            status="submitted",
            prompt=data.prompt,
            remote_task_id=str(remote_id),
            request_data={
                "base_url": base,
                "status_path_template": data.status_path_template,
                "video_url_field": data.video_url_field,
                "status_field": data.status_field,
                "task_id_field": data.task_id_field,
                "confirm_paid": data.confirm_paid
            },
            result_data=result
        )
        db.add(task)
        db.commit()
        db.refresh(task)
        return {"task_id": task.id, "remote_task_id": remote_id,
                "status": task.status, "provider_response": result}
    except Exception as exc:
        raise HTTPException(502, f"视频生成请求失败：{exc}")


@router.post("/video/tasks/{task_id}/refresh")
def refresh_video_task(task_id: int, db: Session = Depends(get_db)):
    task = db.query(MediaTask).filter(MediaTask.id == task_id).first()
    if not task or task.task_type != "video":
        raise HTTPException(404, "视频任务不存在")

    config, key, base = get_provider(
        db, task.provider, bool((task.request_data or {}).get("confirm_paid"))
    )
    settings = task.request_data or {}
    template = settings.get("status_path_template", "/videos/{task_id}")
    path = template.replace("{task_id}", str(task.remote_task_id))
    try:
        result = http_json(base + "/" + path.lstrip("/"), key, "GET")
        status = str(value_at(result, settings.get("status_field", "status")) or "processing")
        task.result_data = result
        task.status = status

        video_url = value_at(result, settings.get("video_url_field", "video_url"))
        if video_url:
            filename = f"video_{task.id}_{uuid.uuid4().hex[:8]}.mp4"
            destination = VIDEO_DIR / filename
            http_download(str(video_url), destination)
            media = MediaFile(
                name=filename, category="video", file_path=str(destination),
                mime_type="video/mp4", file_size=destination.stat().st_size
            )
            db.add(media)
            task.result_path = str(destination)
            task.status = "completed"

        db.commit()
        return {"task_id": task.id, "status": task.status,
                "result_path": task.result_path, "provider_response": result}
    except Exception as exc:
        task.error = str(exc)
        db.commit()
        raise HTTPException(502, f"查询视频任务失败：{exc}")

@router.post("/speech/generate")
def generate_speech(data: SpeechInput, db: Session = Depends(get_db)):
    config, key, base = get_provider(db, data.provider, data.confirm_paid)

    payload = {
        "model": data.model,
        "input": data.text,
        "voice": data.voice,
        "response_format": data.response_format,
        "speed": data.speed
    }
    if data.instructions:
        payload["instructions"] = data.instructions

    filename = f"speech_{uuid.uuid4().hex}.{data.response_format}"
    destination = AUDIO_DIR / filename
    import urllib.request
    import urllib.error

    request = urllib.request.Request(
        base + "/audio/speech",
        data=json.dumps(payload, ensure_ascii=False).encode("utf-8"),
        headers={
            "Authorization": f"Bearer {key}",
            "Content-Type": "application/json"
        },
        method="POST"
    )
    try:
        with urllib.request.urlopen(request, timeout=300) as response:
            with open(destination, "wb") as output:
                while True:
                    chunk = response.read(1024 * 1024)
                    if not chunk:
                        break
                    output.write(chunk)

        media = MediaFile(
            name=filename, category="audio", file_path=str(destination),
            mime_type="audio/" + data.response_format,
            file_size=destination.stat().st_size
        )
        task = MediaTask(
            task_type="speech", provider=data.provider, model=data.model,
            status="completed", prompt=data.text, result_path=str(destination)
        )
        db.add(media)
        db.add(task)
        db.commit()
        return {"status": "completed", "file": filename,
                "path": str(destination), "size": destination.stat().st_size}
    except urllib.error.HTTPError as exc:
        detail = exc.read().decode("utf-8", errors="replace")
        raise HTTPException(502, f"语音 API 返回 HTTP {exc.code}: {detail[:1500]}")
    except Exception as exc:
        raise HTTPException(502, f"语音生成失败：{exc}")


@router.post("/files/upload")
async def upload_media(file: UploadFile = File(...), category: str = "video",
                       db: Session = Depends(get_db)):
    allowed = {"video": VIDEO_DIR, "audio": AUDIO_DIR, "image": STORAGE / "images"}
    if category not in allowed:
        raise HTTPException(400, "不支持的素材类别")

    folder = allowed[category]
    folder.mkdir(parents=True, exist_ok=True)
    suffix = Path(file.filename or "upload.bin").suffix[:20]
    filename = f"{uuid.uuid4().hex}{suffix}"
    destination = folder / filename
    size = 0
    limit = 2 * 1024 * 1024 * 1024

    try:
        with open(destination, "wb") as output:
            while True:
                chunk = await file.read(1024 * 1024)
                if not chunk:
                    break
                size += len(chunk)
                if size > limit:
                    raise HTTPException(413, "单个文件不能超过 2GB")
                output.write(chunk)

        media = MediaFile(
            name=file.filename or filename, category=category,
            file_path=str(destination), mime_type=file.content_type,
            file_size=size
        )
        db.add(media)
        db.commit()
        db.refresh(media)
        return {"id": media.id, "name": media.name, "category": category,
                "size": size, "path": media.file_path}
    except Exception:
        if destination.exists():
            destination.unlink()
        raise

def srt_time(seconds):
    ms = int(round(seconds * 1000))
    hours, ms = divmod(ms, 3600000)
    minutes, ms = divmod(ms, 60000)
    secs, ms = divmod(ms, 1000)
    return f"{hours:02}:{minutes:02}:{secs:02},{ms:03}"


@router.post("/subtitles")
def create_subtitle(data: SubtitleInput, db: Session = Depends(get_db)):
    if not data.cues:
        raise HTTPException(400, "字幕内容不能为空")
    lines = []
    for index, cue in enumerate(data.cues, 1):
        if cue.end <= cue.start:
            raise HTTPException(400, f"第 {index} 条字幕结束时间必须大于开始时间")
        lines.extend([
            str(index),
            f"{srt_time(cue.start)} --> {srt_time(cue.end)}",
            cue.text,
            ""

def decrypt_api_key(value):
    if not value:
        return ""
    key_file = STORAGE / "secrets" / "local.key"
    if not key_file.exists():
        raise RuntimeError("API 密钥加密文件不存在，无法解密已保存的密钥")
    try:
        return Fernet(key_file.read_bytes()).decrypt(value.encode()).decode()
    except Exception as exc:
        raise RuntimeError("API 密钥解密失败，请在 AI 设置中重新保存密钥") from exc


def http_json(url, api_key, method="POST", body=None, timeout=300):
    headers = {
        "Authorization": f"Bearer {api_key}",
        "Content-Type": "application/json",
        "Accept": "application/json",
    }
    data = json.dumps(body or {}, ensure_ascii=False).encode("utf-8")
    request = urllib.request.Request(
        url, data=data if method != "GET" else None,
        headers=headers, method=method
    )
    try:
        with urllib.request.urlopen(request, timeout=timeout) as response:
            raw = response.read()
            content_type = response.headers.get("Content-Type", "")
            if "json" in content_type or raw[:1] in (b"{", b"["):
                return json.loads(raw.decode("utf-8"))
            return {"raw_response": raw.decode("utf-8", errors="replace")}
    except urllib.error.HTTPError as exc:
        detail = exc.read().decode("utf-8", errors="replace")
        raise RuntimeError(f"远程 API 返回 HTTP {exc.code}: {detail[:2000]}") from exc
    except urllib.error.URLError as exc:
        raise RuntimeError(f"无法连接远程 API: {exc}") from exc


def http_download(url, destination, api_key=None):
    headers = {}
    if api_key:
        headers["Authorization"] = f"Bearer {api_key}"
    request = urllib.request.Request(url, headers=headers)
    try:
        with urllib.request.urlopen(request, timeout=600) as response:
            with open(destination, "wb") as output:
                while True:
                    chunk = response.read(1024 * 1024)
                    if not chunk:
                        break
                    output.write(chunk)
    except Exception:
        if os.path.exists(destination):
            os.remove(destination)
        raise


def safe_storage_file(path):
    candidate = Path(path).resolve()
    root = STORAGE.resolve()
    if candidate != root and root not in candidate.parents:
        raise ValueError("文件路径超出素材存储目录")
    if not candidate.is_file():
        raise FileNotFoundError("指定的素材文件不存在")
    return candidate


def run_ffmpeg(args):
    executable = os.environ.get("FFMPEG_PATH", "ffmpeg")
    command = [executable, "-y"] + [str(item) for item in args]
    try:
        result = subprocess.run(
            command, capture_output=True, text=True, timeout=7200
        )
    except FileNotFoundError as exc:
        raise RuntimeError("未找到 FFmpeg，请先安装 FFmpeg 并配置 PATH") from exc
    if result.returncode != 0:
        raise RuntimeError("FFmpeg 执行失败：" + result.stderr[-3000:])
    return result


def probe_media(path):
    executable = os.environ.get("FFPROBE_PATH", "ffprobe")
    command = [
        executable, "-v", "quiet", "-print_format", "json",
        "-show_format", "-show_streams", str(path)
    ]
    try:
        result = subprocess.run(
            command, capture_output=True, text=True, timeout=120
        )
    except FileNotFoundError as exc:
        raise RuntimeError("未找到 FFprobe，请安装 FFmpeg 完整版本") from exc
    if result.returncode != 0:
        raise RuntimeError(result.stderr[-2000:])
    return json.loads(result.stdout)

def call_ai_for_edit_plan(db, provider, model, instruction, source_files,
                          confirm_paid=False):
    config, key, base = get_ai_config(db, provider)

    if bool(getattr(config, "paid", True)) and not confirm_paid:
        raise RuntimeError("本次 AI 剪辑分析可能产生费用，请先明确确认")

    system_prompt = """
你是专业影视剪辑师。你必须把用户的剪辑要求转换成严格 JSON。
只能使用用户提供的视频文件路径，不得编造路径。
只返回 JSON，不要 Markdown，不要解释。

JSON 格式：
{
  "title": "剪辑项目名称",
  "summary": "剪辑方案说明",
  "segments": [
    {
      "file_path": "必须是输入文件中的路径",

  "height": 1080,
  "subtitle_text": "",

规则：
1. start 和 end 是视频原始时间，单位秒。
2. end 必须大于 start。
3. 片段必须来自用户提供的文件。
4. 按用户要求排序 segments。
5. 不要编造画面内容；无法判断的内容写入 notes。
6. 用户没有要求删除内容时，尽量保留相关内容。
7. 不要在 JSON 外输出任何文字。
"""

    user_prompt = (
        "用户剪辑要求：\n" + instruction +
        "\n\n允许使用的视频文件：\n" +
        json.dumps(source_files, ensure_ascii=False)
    )

    payload = {
        "model": model,
        "temperature": 0.2,
        "messages": [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_prompt}
        ]
    }

    request = urllib.request.Request(
        base + "/chat/completions",
        data=json.dumps(payload, ensure_ascii=False).encode("utf-8"),
        headers={
            "Authorization": f"Bearer {key}",
            "Content-Type": "application/json"

def validate_plan(plan, allowed_files):
    if not plan.get("segments"):
        raise ValueError("剪辑方案没有可执行的视频片段")

    allowed = {str(Path(x).resolve()) for x in allowed_files}
    validated = []

    for index, segment in enumerate(plan["segments"], 1):
        if not isinstance(segment, dict):
            raise ValueError(f"第 {index} 个片段格式错误")

        raw_path = segment.get("file_path")
        if not raw_path:
            raise ValueError(f"第 {index} 个片段缺少视频路径")

        path = safe_storage_file(raw_path)
        if str(path.resolve()) not in allowed:
            raise ValueError(f"第 {index} 个片段使用了未授权的输入文件")

        start = float(segment.get("start", 0))
        end = float(segment.get("end", 0))

        if start < 0 or end <= start:
            raise ValueError(f"第 {index} 个片段时间范围无效")

        if end - start > 7200:
            raise ValueError(f"第 {index} 个片段时长超过限制")

        validated.append({
            "file_path": str(path),
            "start": start,
            "end": end,
            "description": str(segment.get("description", ""))[:1000]
        })

    width = int(plan.get("width", 1920))
    height = int(plan.get("height", 1080))

    if not (256 <= width <= 3840 and 256 <= height <= 3840):
        raise ValueError("输出分辨率超出允许范围")

    return {
        "title": str(plan.get("title", "视频剪辑"))[:300],
        "summary": str(plan.get("summary", ""))[:5000],
        "segments": validated,
        "width": width,
        "height": height,
        "subtitle_text": str(plan.get("subtitle_text", ""))[:10000],
        "notes": plan.get("notes", [])
    }


def execute_edit_plan(plan, subtitle_path=None, audio_path=None):
    work = EDIT_DIR / uuid.uuid4().hex
    work.mkdir(parents=True, exist_ok=True)

    clips = []

    for index, segment in enumerate(plan["segments"]):
        output = work / f"segment_{index:04d}.mp4"
        duration = segment["end"] - segment["start"]

        args = [
            "-ss", str(segment["start"]),
            "-i", segment["file_path"],
            "-t", str(duration),
            "-vf",
            f"scale={plan['width']}:{plan['height']}:force_original_aspect_ratio=decrease,"
            f"pad={plan['width']}:{plan['height']}:(ow-iw)/2:(oh-ih)/2",
            "-c:v", "libx264",
            "-preset", "veryfast",
            "-crf", "21",
            "-c:a", "aac",
            "-ar", "48000",
            "-ac", "2",
            "-avoid_negative_ts", "make_zero",
            "-movflags", "+faststart",
            str(output)
        ]
        run_ffmpeg(args)
        clips.append(output)

    concat_file = work / "concat.txt"
    concat_file.write_text(
        "".join("file '" + str(x).replace("\\", "/").replace("'", "'\\''") + "'\n"
                for x in clips),
        encoding="utf-8"
    )

    output = VIDEO_DIR / f"ai_edit_{uuid.uuid4().hex}.mp4"

    args = [
        "-f", "concat",
        "-safe", "0",
        "-i", str(concat_file)
    ]

    audio = safe_storage_file(audio_path) if audio_path else None
    if audio:
        args += ["-i", str(audio)]

    subtitle = safe_storage_file(subtitle_path) if subtitle_path else None

    if subtitle:
        escaped = str(subtitle).replace("\\", "/").replace(":", "\\:")
        args += ["-vf", f"subtitles='{escaped}'"]

    if audio:
        args += ["-map", "0:v:0", "-map", "1:a:0", "-shortest"]
    else:
        args += ["-map", "0:v:0", "-map", "0:a?"]

    args += [
        "-c:v", "libx264",
        "-preset", "medium",
        "-crf", "20",
        "-c:a", "aac",
        "-movflags", "+faststart",
        str(output)
    ]

    run_ffmpeg(args)

    return str(output), output.stat().st_size

========== apps\web\src\app\novels\page.tsx ==========
"use client";

import { useEffect, useRef, useState } from "react";

const API = process.env.NEXT_PUBLIC_API_BASE_URL || "http://127.0.0.1:8000";

type Novel = {
  id: number;
  title: string;
  description: string;
  genre: string;
  status: string;
  chapter_count: number;
  word_count: number;
};

type Chapter = {
  id: number;
  chapter_number: number;
  title: string;
  content: string;
  word_count: number;
};

type AnalysisItem = {
  id: number;
  category: string;
  name: string;
  description: string;
  confirmed: boolean;
};

const categoryNames: Record<string, string> = {
  character: "人物",
  scene: "场景",
  prop: "道具",
  event: "故事事件",
  relationship: "人物关系",
  world: "世界观",
};

export default function NovelsPage() {
  const [novels, setNovels] = useState<Novel[]>([]);
  const [selected, setSelected] = useState<Novel | null>(null);
  const [chapters, setChapters] = useState<Chapter[]>([]);
  const [currentChapter, setCurrentChapter] = useState<Chapter | null>(null);
  const [analysis, setAnalysis] = useState<AnalysisItem[]>([]);
  const [title, setTitle] = useState("");
  const [genre, setGenre] = useState("奇幻");
  const [loading, setLoading] = useState(false);
  const [message, setMessage] = useState("");
  const fileRef = useRef<HTMLInputElement>(null);

  async function request(path: string, options?: RequestInit) {
    const response = await fetch(`${API}${path}`, options);
    const data = await response.json().catch(() => ({}));
    if (!response.ok) throw new Error(data.detail || "请求失败");
    return data;
  }

  async function loadNovels() {
    try {
      const data = await request("/api/novels");
      setNovels(data.items || []);
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "无法连接后端");
    }
  }

  async function openNovel(novel: Novel) {
    setSelected(novel);
    setCurrentChapter(null);
    try {
      const [chapterData, analysisData] = await Promise.all([
        request(`/api/novels/${novel.id}/chapters`),
        request(`/api/novels/${novel.id}/analysis`),
      ]);
      setChapters(chapterData.items || []);
      setAnalysis(analysisData.items || []);
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "读取失败");
    }
  }

  useEffect(() => {
    loadNovels();
  }, []);

  async function createNovel() {
    if (!title.trim()) {
      setMessage("请先填写小说名称");
      return;
    }
    setLoading(true);
    try {
      const item = await request("/api/novels", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ title: title.trim(), genre }),
      });
      setTitle("");
      await loadNovels();
      await openNovel(item);
      setMessage("小说项目创建成功");
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "创建失败");
    } finally {
      setLoading(false);
    }
  }

  async function importTxt() {
    const file = fileRef.current?.files?.[0];
    if (!selected || !file) {
      setMessage("请先选择小说，再选择 TXT 文件");
      return;
    }
    setLoading(true);
    try {
      const form = new FormData();
      form.append("file", file);
      const data = await request(`/api/novels/${selected.id}/import-txt`, {
        method: "POST",
        body: form,
      });
      setChapters(data.chapters || []);
      setCurrentChapter(null);
      await loadNovels();
      setMessage(`导入成功：${data.chapter_count} 章，共 ${data.word_count} 字`);
      if (fileRef.current) fileRef.current.value = "";
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "导入失败");
    } finally {
      setLoading(false);
    }
  }

  async function saveChapter() {
    if (!currentChapter) return;
    setLoading(true);
    try {
      await request(`/api/chapters/${currentChapter.id}`, {
        method: "PUT",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          title: currentChapter.title,
          content: currentChapter.content,
        }),
      });
      await openNovel(selected!);
      setMessage("章节保存成功");
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "保存失败");
    } finally {
      setLoading(false);
    }
  }

  async function analyze() {
    if (!selected) return;
    setLoading(true);
    try {
      const data = await request(`/api/novels/${selected.id}/analyze`, {
        method: "POST",
      });
      const result = await request(`/api/novels/${selected.id}/analysis`);
      setAnalysis(result.items || []);
      setMessage(data.message || "初步解析完成");
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "解析失败");
    } finally {
      setLoading(false);
    }
  }

  async function confirmAnalysis() {
    if (!selected) return;
    try {
      await request(`/api/novels/${selected.id}/analysis/confirm`, {
        method: "POST",
      });
      const result = await request(`/api/novels/${selected.id}/analysis`);
      setAnalysis(result.items || []);
      setMessage("解析结果已确认并保存");
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "确认失败");
    }
  }

  async function removeNovel() {
    if (!selected || !window.confirm("确定删除这部小说及其章节和解析数据吗？")) return;
    try {
      await request(`/api/novels/${selected.id}`, { method: "DELETE" });
      setSelected(null);
      setChapters([]);
      setAnalysis([]);
      await loadNovels();
      setMessage("小说已删除");
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "删除失败");
    }
  }

  async function editAnalysis(item: AnalysisItem) {
    const next = window.prompt("修改名称", item.name);
    if (next === null || !next.trim()) return;
    try {
      await request(`/api/analysis/${item.id}`, {
        method: "PUT",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ name: next.trim() }),
      });
      const result = await request(`/api/novels/${selected!.id}/analysis`);
      setAnalysis(result.items || []);
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "修改失败");
    }
  }

  async function deleteAnalysis(id: number) {
    try {
      await request(`/api/analysis/${id}`, { method: "DELETE" });
      setAnalysis((items) => items.filter((item) => item.id !== id));
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "删除失败");
    }
  }

  return (
    <main className="cosmic min-h-screen text-white">
      <div className="nebula nebula-one" />
      <div className="nebula nebula-two" />
      <div className="main-layout relative z-10 min-h-screen p-5 lg:p-8">
        <header className="mb-7 flex flex-wrap items-center justify-between gap-4">
          <div>
            <a href="/" className="text-sm text-purple-300 hover:text-white">← 返回首页</a>
            <h1 className="mt-3 text-3xl font-black">📚 小说创作工作室</h1>
            <p className="mt-2 text-sm text-purple-200">小说管理、TXT导入、章节编辑与内容解析</p>
          </div>
          <div className="rounded-full border border-purple-400/30 bg-purple-950/60 px-4 py-2 text-sm">
            本地数据库 · 自动保存项目数据
          </div>
        </header>

        {message && (
          <div className="mb-5 rounded-xl border border-purple-400/30 bg-purple-950/70 p-3 text-sm text-purple-100">
            {message}
          </div>
        )}

        <div className="grid grid-cols-1 gap-5 xl:grid-cols-[340px_minmax(0,1fr)]">
          <aside className="rounded-2xl border border-purple-400/20 bg-[#160d2acc] p-5">
            <h2 className="mb-4 text-lg font-bold">小说项目</h2>
            <div className="space-y-3">
              <input
                value={title}
                onChange={(e) => setTitle(e.target.value)}
                placeholder="输入小说名称"
                className="w-full rounded-xl border border-purple-400/30 bg-black/30 px-4 py-3 outline-none focus:border-purple-300"
              />
              <select
                value={genre}
                onChange={(e) => setGenre(e.target.value)}
                className="w-full rounded-xl border border-purple-400/30 bg-[#211338] px-4 py-3"
              >
                {["奇幻", "玄幻", "都市", "言情", "科幻", "悬疑", "历史", "武侠", "其他"].map((x) => (
                  <option key={x}>{x}</option>
                ))}
              </select>
              <button
                onClick={createNovel}
                disabled={loading}
                className="w-full rounded-xl bg-gradient-to-r from-purple-600 to-fuchsia-600 px-4 py-3 font-bold disabled:opacity-50"
              >
                ＋ 新建小说项目
              </button>
            </div>

            <div className="mb-3 mt-7 flex items-center justify-between">
              <h3 className="font-semibold">我的小说</h3>
              <button onClick={loadNovels} className="text-xs text-purple-300">刷新</button>
            </div>
            <div className="space-y-2">
              {novels.map((novel) => (
                <button
                  key={novel.id}
                  onClick={() => openNovel(novel)}
                  className={`w-full rounded-xl border p-3 text-left transition ${
                    selected?.id === novel.id
                      ? "border-purple-300 bg-purple-700/30"
                      : "border-purple-400/15 bg-black/20 hover:bg-purple-900/40"
                  }`}
                >
                  <div className="font-semibold">{novel.title}</div>
                  <div className="mt-2 text-xs text-purple-300">
                    {novel.genre} · {novel.chapter_count} 章 · {novel.word_count} 字
                  </div>
                </button>
              ))}
              {novels.length === 0 && (
                <p className="rounded-xl bg-black/20 p-4 text-sm text-purple-300">还没有小说，先创建一个项目。</p>
              )}
            </div>
          </aside>

          <section className="min-w-0 space-y-5">
            {!selected ? (
              <div className="flex min-h-[430px] flex-col items-center justify-center rounded-2xl border border-purple-400/20 bg-[#160d2acc] p-8 text-center">
                <div className="text-6xl">📖</div>
                <h2 className="mt-5 text-2xl font-bold">开始你的小说创作</h2>
                <p className="mt-3 max-w-lg text-sm leading-7 text-purple-200">
                  创建小说项目后，可以导入 TXT、管理章节、编辑内容，并进行人物、场景、道具和剧情线索的初步识别。
                </p>
              </div>
            ) : (
              <>
                <div className="rounded-2xl border border-purple-400/20 bg-[#160d2acc] p-5">
                  <div className="flex flex-wrap items-center justify-between gap-3">
                    <div>
                      <h2 className="text-2xl font-bold">{selected.title}</h2>

      await request(`/api/story/episodes/${episode.id}`, {
        method: "PUT",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(patch),
      });
      if (projectId) await loadProject(projectId);
      setMessage("分集信息已保存");
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "保存失败");
    }
  }

  async function splitEpisode(id: number) {
    try {
      await request(`/api/story/episodes/${id}/split`, { method: "POST" });
      if (projectId) await loadProject(projectId);
      setMessage("分集已拆分");
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "拆分失败");
    }
  }

  async function mergeEpisodes() {
    if (!projectId || episodes.length < 2) return;
    const ids = episodes.slice(0, 2).map(x => x.id);
    try {
      await request(`/api/story/projects/${projectId}/merge`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ episode_ids: ids, title: "合并分集" }),
      });
      await loadProject(projectId);
      setMessage("前两集已合并");
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "合并失败");
    }
  }

  async function moveEpisode(index: number, direction: number) {
    if (!projectId) return;
    const next = [...episodes];
    const target = index + direction;
    if (target < 0 || target >= next.length) return;
    [next[index], next[target]] = [next[target], next[index]];
    try {
      await request(`/api/story/projects/${projectId}/reorder`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ episode_ids: next.map(x => x.id) }),
      });
      await loadProject(projectId);
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "排序失败");
    }
  }

  async function addShot() {
    if (!episodeId) return setMessage("请先选择分集");
    try {
      await request(`/api/story/episodes/${episodeId}/shots`, {
        method: "POST",

        body: JSON.stringify({
          ...shotForm,

        }),
      });

  async function deleteShot(id: number) {

    } catch (error) {

        headers: { "Content-Type": "application/json" },

      });
      setMessage(`已将素材关联到镜头 #${shot.shot_number}`);
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "素材关联失败");
    }
  }

  return (

    setMessage("正在调用 AI 分析剪辑要求……");
    try {
      const response = await fetch(`${API}/api/video-editor/jobs`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          provider, model, instruction,
          source_paths: selected,
          confirm_paid: confirmPaid
        })
      });

      await loadAll();
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "AI 分析失败");
    } finally {
      setBusy(false);
    }
  }

  async function executeJob(id: number) {
    setBusy(true);
    setMessage("FFmpeg 正在执行视频剪辑，请等待……");
    try {
      const response = await fetch(`${API}/api/video-editor/jobs/${id}/execute`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          subtitle_path: subtitlePath || null,
          audio_path: audioPath || null
        })
      });
      const data = await response.json();
      if (!response.ok) throw new Error(data.detail || "剪辑执行失败");
      setMessage(`视频导出完成：${data.output_path}`);
      await loadAll();
    } catch (error) {
      setMessage(error instanceof Error ? error.message : "剪辑失败");
      await loadAll();
    } finally {
      setBusy(false);
    }
  }

  function toggleFile(path: string) {
    setSelected(current =>
      current.includes(path)
        ? current.filter(x => x !== path)
        : [...current, path]
    );
  }

  function formatSize(size: number) {
    return size >= 1024 ** 3
      ? `${(size / 1024 ** 3).toFixed(2)} GB`
      : `${(size / 1024 ** 2).toFixed(1)} MB`;
  }

      <div className="mx-auto max-w-7xl space-y-7">
        <header>
          <h1 className="text-3xl font-bold">AI 视频剪辑中心</h1>
          <p className="mt-2 text-white/60">
            用文字描述剪辑要求，让 AI 规划，再由本地 FFmpeg 执行
          </p>
        </header>

        {message && (
          <div className="rounded-xl border border-purple-400/30 bg-purple-500/10 p-4">
            {message}
          </div>

            <label className="block cursor-pointer rounded-xl border border-dashed border-purple-300/40 p-8 text-center hover:bg-purple-500/10">

export default function AICenterPage() {
  const [providers, setProviders] = useState<Provider[]>([]);
  const [network, setNetwork] = useState(false);
  const [paid, setPaid] = useState(false);
  const [selected, setSelected] = useState(presets[0]);
  const [apiKey, setApiKey] = useState("");
  const [baseUrl, setBaseUrl] = useState(presets[0].url);
  const [model, setModel] = useState(presets[0].model);
  const [enabled, setEnabled] = useState(false);
  const [logs, setLogs] = useState<any[]>([]);
  const [message, setMessage] = useState("");

  async function api(path: string, options?: RequestInit) {
    const response = await fetch(`${API}${path}`, options);
    const data = await response.json().catch(() => ({}));
    if (!response.ok) throw new Error(data.detail || "请求失败");
    return data;
  }

  async function refresh() {
    try {
      const [p, r, l] = await Promise.all([
        api("/api/ai/providers"),
        api("/api/ai/runtime"),
        api("/api/ai/usage-logs"),
      ]);
      setProviders(p.items || []);
      setNetwork(r.network_enabled);
      setPaid(r.paid_api_allowed);
      setLogs(l.items || []);
    } catch (e) {
      setMessage(e instanceof Error ? e.message : "读取失败");
    }
  }

  useEffect(() => { refresh(); }, []);

  function choose(item: typeof presets[number]) {
    setSelected(item);
    setBaseUrl(item.url);
    setModel(item.model);
    setApiKey("");
    const old = providers.find((x) => x.provider === item.id);
    setEnabled(old?.enabled || false);
  }

  async function save() {
    try {
      await api("/api/ai/providers", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          provider: selected.id,
          display_name: selected.name,
          base_url: baseUrl,
          model,
          api_key: apiKey,
          enabled,
          paid: selected.paid,
          purpose: selected.purpose,
        }),
      });
      setApiKey("");
      await refresh();
      setMessage(`${selected.name} 配置已保存`);
    } catch (e) {
      setMessage(e instanceof Error ? e.message : "保存失败");
    }
  }

  async function saveRuntime(nextNetwork: boolean, nextPaid: boolean) {
    try {
      await api("/api/ai/runtime", {
        method: "PUT",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          network_enabled: nextNetwork,
          paid_api_allowed: nextPaid,
        }),
      });
      setNetwork(nextNetwork);
      setPaid(nextPaid);
      setMessage("调用权限设置已保存");
    } catch (e) {
      setMessage(e instanceof Error ? e.message : "设置失败");
    }
  }

  async function removeProvider() {
    if (!confirm(`确定删除 ${selected.name} 配置吗？`)) return;
    try {
      await api(`/api/ai/providers/${selected.id}`, { method: "DELETE" });
      await refresh();
      setMessage("配置已删除");
    } catch (e) {
      setMessage(e instanceof Error ? e.message : "删除失败");
    }
  }

  return (
    <main className="cosmic min-h-screen text-white">
      <div className="nebula nebula-one" />
      <div className="nebula nebula-two" />
      <div className="relative z-10 min-h-screen p-5 lg:p-8">
        <header className="mb-7">
          <a href="/" className="text-sm text-purple-300">← 返回首页</a>
          <h1 className="mt-3 text-3xl font-black">⚙️ AI 模型管理中心</h1>
          <p className="mt-2 text-sm text-purple-200">配置 AI 服务、控制网络权限、管理调用记录</p>
        </header>

        {message && <div className="mb-5 rounded-xl border border-purple-400/30 bg-purple-950/70 p-3 text-sm">{message}</div>}

        <section className="mb-5 grid grid-cols-1 gap-4 md:grid-cols-2">
          <div className="rounded-2xl border border-purple-400/20 bg-[#160d2acc] p-5">
            <h2 className="font-bold">网络调用总开关</h2>
            <p className="my-2 text-xs text-purple-300">关闭后，所有外部 AI 请求都会被阻止。</p>
            <button onClick={() => saveRuntime(!network, paid)} className={`rounded-lg px-5 py-2 text-sm font-bold ${network ? "bg-emerald-700" : "bg-gray-700"}`}>
              {network ? "已开启" : "已关闭"}
            </button>
          </div>
          <div className="rounded-2xl border border-purple-400/20 bg-[#160d2acc] p-5">
            <h2 className="font-bold">付费 API 总授权</h2>
            <p className="my-2 text-xs text-purple-300">开启仅表示允许付费请求，还必须在具体请求时再次确认。</p>
            <button onClick={() => saveRuntime(network, !paid)} className={`rounded-lg px-5 py-2 text-sm font-bold ${paid ? "bg-amber-700" : "bg-gray-700"}`}>
              {paid ? "已授权" : "未授权"}
            </button>
          </div>
        </section>

        <section className="grid grid-cols-1 gap-5 xl:grid-cols-[300px_minmax(0,1fr)]">
          <aside className="rounded-2xl border border-purple-400/20 bg-[#160d2acc] p-5">
            <h2 className="mb-4 text-lg font-bold">AI 服务商</h2>
            <div className="space-y-2">
              {presets.map((p) => {
                const saved = providers.find((x) => x.provider === p.id);

                <label className="text-sm text-purple-200">API 地址
                  <input value={baseUrl} onChange={(e) => setBaseUrl(e.target.value)} className="mt-2 w-full rounded-lg border border-purple-400/30 bg-black/30 p-3 text-white" />
                </label>
                <label className="text-sm text-purple-200">模型名称
                  <input value={model} onChange={(e) => setModel(e.target.value)} className="mt-2 w-full rounded-lg border border-purple-400/30 bg-black/30 p-3 text-white" />
